## Examples

```tsx
// Card border
<div sz={{
  border: 1,
  borderColor: 'gray-200',
  rounded: 'lg',
  p: 6,
}} />

// Focus ring on button
<button sz={{
  focusVisible: {
    outline: 'none',
    ring: 2,
    ringColor: 'blue-500',
    ringOffset: 2,
  },
}} />

// Left accent border (callout style)
<div sz={{
  border: 1,
  borderL: '3px',
  borderColor: 'gray-200',
  borderLColor: 'blue-500',
  pl: 4,
  rounded: 'sm',
}} />

// Divider between list items
<ul sz={{ divideY: 1, divideColor: 'gray-100' }}>
  <li sz={{ py: 3 }}>Item 1</li>
  <li sz={{ py: 3 }}>Item 2</li>
</ul>
```
