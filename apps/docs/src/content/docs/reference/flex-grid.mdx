## Examples

```tsx
// Centered hero section
<div sz={{
  flex: true,
  items: 'center',
  justify: 'center',
  minH: 'screen',
}} />

// Responsive grid
<div sz={{
  grid: true,
  gridCols: 1,
  gap: 6,
  sm: { gridCols: 2 },
  lg: { gridCols: 3 },
}} />

// Navigation bar
<nav sz={{
  flex: true,
  items: 'center',
  justify: 'between',
  px: 6,
  py: 4,
}}>
  <div sz={{ flex: true, items: 'center', gap: 8 }}>
    <Logo />
    <Links />
  </div>
  <Actions />
</nav>

// Sidebar layout
<div sz={{ flex: true, gap: 8, minH: 'screen' }}>
  <aside sz={{ w: 64, shrink: 0 }} />
  <main sz={{ grow: true }} />
</div>
```
